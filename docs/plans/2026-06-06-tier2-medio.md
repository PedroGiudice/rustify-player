O item 2.1 usa apenas `libCreateStation` com seed; `lib_find_similar` é mencionado no triage como "opcional". O plano não inclui `lib_find_similar` como obrigatório — o CTA cria a station com seed, o engine já popula via `lib_play_station` que usa `lib_find_similar` internamente.

**Step 4.2 — Implementar: corrigir contrato u64/string no Rust**

Editar `src-tauri/src/lib.rs`, função `lib_create_station` (linha 3071):

Mudar o parâmetro de `seed_track_ids: Option<Vec<u64>>` para `seed_track_ids: Option<Vec<String>>`:

```rust
#[tauri::command]
fn lib_create_station(
    lib: State<Library>,
    name: String,
    kind: String,
    seed_track_ids: Option<Vec<String>>,
    query: Option<String>,
    icon: Option<String>,
    tone: Option<String>,
    desc: Option<String>,
) -> Result<Station, String> {
    let station_kind = match kind.as_str() {
        "mood" => StationKind::Mood,
        _ => StationKind::Seed,
    };
    let slug: String = name
        .chars()
        .map(|c| if c.is_alphanumeric() { c.to_ascii_lowercase() } else { '-' })
        .collect::<String>()
        .trim_matches('-')
        .to_string();
    let ts = now_unix();
    let id = format!("{slug}-{ts}");

    // Converte Vec<String> -> Vec<u64> via parse_id (mesmo helper do resto do codebase)
    let ids_u64: Vec<u64> = seed_track_ids
        .unwrap_or_default()
        .into_iter()
        .filter_map(|s| parse_id(&s).ok())
        .collect();

    let station = Station {
        id,
        name,
        icon: icon.unwrap_or_else(|| "lucide:radio".to_string()),
        tone: tone.unwrap_or_else(|| "tone-lavender".to_string()),
        desc: desc.unwrap_or_default(),
        kind: station_kind,
        seed_track_ids: ids_u64,
        query,
        stats: StationStats::default(),
    };
    write_station(&lib.data_dir, &station)?;
    Ok(station)
}
```

```typescript
/** Move o pin de `fromName` para logo antes de `toName`.
    Se `toName` for null, move para o final. */
export function reorderPins(fromName: string, toName: string | null) {
  const current = pins();
  const fromIdx = current.indexOf(fromName);
  if (fromIdx < 0) return; // item não está pinado
  const next = current.filter((n) => n !== fromName);
  if (toName === null) {
    next.push(fromName);
  } else {
    const toIdx = next.indexOf(toName);
    if (toIdx < 0) { next.push(fromName); }
    else { next.splice(toIdx, 0, fromName); }
  }
  setPins(next);
  save(next);
}
```

```rust
/// Resultado de um .m3u para o frontend
#[derive(Serialize, Clone)]
struct M3uPlaylist {
    name: String,        // nome do arquivo sem extensão
    path: String,        // path absoluto
    track_count: usize,  // número de linhas não-comentário
}

/// Diretório onde ficam os .m3u gerenciados pelo app
fn m3u_dir(music_root: &std::path::Path) -> std::path::PathBuf {
    music_root.join("playlists")
}

/// Lista todas as playlists .m3u em <music_root>/playlists/
#[tauri::command]
fn lib_list_m3u_playlists(lib: State<Library>) -> Result<Vec<M3uPlaylist>, String> {
    let dir = m3u_dir(&lib.music_root);
    if !dir.exists() {
        return Ok(vec![]);
    }
    let mut result = Vec::new();
    let entries = std::fs::read_dir(&dir).map_err(|e| e.to_string())?;
    for entry in entries.flatten() {
        let path = entry.path();
        if path.extension().and_then(|e| e.to_str()) != Some("m3u") {
            continue;
        }
        let name = path
            .file_stem()
            .and_then(|s| s.to_str())
            .unwrap_or("sem nome")
            .to_string();
        let content = std::fs::read_to_string(&path).unwrap_or_default();
        let track_count = content
            .lines()
            .filter(|l| !l.trim().is_empty() && !l.starts_with('#'))
            .count();
        result.push(M3uPlaylist {
            name,
            path: path.to_string_lossy().to_string(),
            track_count,
        });
    }
    result.sort_by(|a, b| a.name.cmp(&b.name));
    Ok(result)
}
```

```rust
/// Cria um .m3u vazio com o nome dado em <music_root>/playlists/<name>.m3u
/// Falha se o arquivo já existir (nomes únicos).
#[tauri::command]
fn lib_create_m3u_playlist(lib: State<Library>, name: String) -> Result<M3uPlaylist, String> {
    let dir = m3u_dir(&lib.music_root);
    std::fs::create_dir_all(&dir).map_err(|e| e.to_string())?;

    // Sanitiza o nome: alfanumérico + espaço + hífen/underscore
    let safe_name: String = name
        .chars()
        .map(|c| if c.is_alphanumeric() || " -_".contains(c) { c } else { '_' })
        .collect();
    if safe_name.trim().is_empty() {
        return Err("nome inválido".to_string());
    }

    let path = dir.join(format!("{safe_name}.m3u"));
    if path.exists() {
        return Err(format!("playlist '{safe_name}' já existe"));
    }

    // Escreve cabeçalho mínimo M3U
    let header = "#EXTM3U\n";
    std::fs::write(&path, header).map_err(|e| e.to_string())?;

    Ok(M3uPlaylist {
        name: safe_name,
        path: path.to_string_lossy().to_string(),
        track_count: 0,
    })
}
```

```rust
/// Acrescenta uma linha com o path da track ao .m3u indicado.
/// Idempotente: não duplica se o path já estiver no arquivo.
#[tauri::command]
fn lib_append_to_m3u(
    _lib: State<Library>,
    m3u_path: String,
    track_path: String,
) -> Result<(), String> {
    let content = std::fs::read_to_string(&m3u_path).unwrap_or_default();

    // Idempotência: não duplicar
    if content.lines().any(|l| l.trim() == track_path.trim()) {
        return Ok(());
    }

    // Append: adiciona nova linha ao final
    let mut file = std::fs::OpenOptions::new()
        .append(true)
        .open(&m3u_path)
        .map_err(|e| e.to_string())?;
    use std::io::Write;
    writeln!(file, "{track_path}").map_err(|e| e.to_string())?;
    Ok(())
}
```
